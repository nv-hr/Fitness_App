    **3. `fuzzyMatchFoodName(name, dbFoods)`**
    Per CONTEXT.md D-02: exact → case-insensitive → substring → Levenshtein ≤ 3 → remove.
    
    Input: `name` (string from LLM), `dbFoods` (array of `{ id, name, calories_per_100g, category }`).
    Returns: `{ matched: boolean, food: object|null, matchType: string }`
    
    Steps:
    1. **Exact match** — `dbFoods.find(f => f.name === name)` → return `matchType: 'exact'`
    2. **Case-insensitive match** — `dbFoods.find(f => f.name.toLowerCase() === name.trim().toLowerCase())` → return `matchType: 'case-insensitive'`
    3. **Substring match** — `dbFoods.find(f => f.name.toLowerCase().includes(name.toLowerCase()) || name.toLowerCase().includes(f.name.toLowerCase()))` → return `matchType: 'substring'`
    4. **Levenshtein ≤ 3** — compute distance for each dbFood name against the input name, find minimum. If `bestDistance <= 3`, return `matchType: 'levenshtein'`
    5. **No match** — return `{ matched: false, food: null, matchType: 'none' }`
    
    Log warnings via `console.warn` for non-exact matches with the pattern: `[MealPlan] Fuzzy match: "{input}" → "{matchedName}" ({matchType})`

    ```javascript
    export async function generateMealPlan(deps) {
      const { userId, weekStart, getProfile, getAllFoods, getLogHistory } = deps;

      // Step 1: Check in-memory cache first
      const cached = getCachedPlan(userId, weekStart);
      if (cached) {
        return { plan: cached, fromCache: true, status: 'active' };
      }

      // Step 2: Fetch dependencies in parallel
      let profile, dbFoods, recentLogs;
      try {
        [profile, dbFoods, recentLogs] = await Promise.all([
          getProfile(userId),
          getAllFoods(userId),
          getLogHistory(userId, 7),  // last 7 days per D-01
        ]);
      } catch (err) {
        console.error('[MealPlan] Failed to fetch user data:', err.message);
        // Fetch basic food list for fallback (can't use profile if that failed)
        try { dbFoods = await getAllFoods(userId); } catch { dbFoods = []; }
        const fallback = await generateFallbackMealPlan(2000, weekStart, dbFoods);
        return { plan: fallback, fromCache: false, status: 'fallback' };
      }

      // Guard: no profile
      if (!profile || !profile.profile) {
        const fallback = await generateFallbackMealPlan(2000, weekStart, dbFoods);
        return { plan: fallback, fromCache: false, status: 'fallback' };
      }

      // Step 3: Compute calorie target
      const calorieTarget = profile.calorieTarget || 2000;
      const userProfile = profile.profile;

      // Step 4: Build prompt
      const prompt = buildMealPlanPrompt(userProfile, dbFoods, recentLogs, weekStart, calorieTarget);

      // Step 5: Attempt loop (max 2 attempts per D-03)
      let plan;
      let attempt = 0;
      const maxAttempts = 2;
      let needsCorrection = false;

      while (attempt < maxAttempts) {
        attempt++;

        try {
          if (!needsCorrection) {
            plan = await callLlmApi(prompt);
          } else {
            // Use correction prompt
            const correctionPrompt = prompt + '\n\n' + buildCorrectionPrompt(lastErrors);
            plan = await callLlmApi(correctionPrompt);
          }
        } catch (err) {
          console.error(`[MealPlan] LLM API call attempt ${attempt} failed:`, err.message);
          if (attempt >= maxAttempts) {
            const fallback = await generateFallbackMealPlan(calorieTarget, weekStart, dbFoods);
            return { plan: fallback, fromCache: false, status: 'fallback' };
          }
          continue;
        }

        // Step 6: Validate structure
        const structureCheck = validateMealPlanStructure(plan, weekStart, calorieTarget);
        if (!structureCheck.valid) {
          console.warn(`[MealPlan] Structure validation failed (attempt ${attempt}):`, structureCheck.errors.join('; '));
          if (attempt < maxAttempts) {
            needsCorrection = true;
            lastErrors = structureCheck.errors;
            continue;
          }
          break;
        }

        // Step 7: Validate and fix names/calories
        const nameCheck = validateAndFixMealPlan(plan, dbFoods);
        if (!nameCheck.valid) {
          console.warn(`[MealPlan] Name validation failed (attempt ${attempt}):`, nameCheck.errors.join('; '));
          if (attempt < maxAttempts) {
            needsCorrection = true;
            lastErrors = nameCheck.errors;
            continue;
          }
          break;
        }

        // Step 8: Success — persist and cache
        plan = nameCheck.plan;
        plan.calorie_target = calorieTarget;
        plan.generated_at = new Date().toISOString();
        plan.llm_model = CONFIG?.model || 'unknown';

        // Persist to DB
        try {
          await upsertPlan(userId, weekStart, plan, 'active');
        } catch (err) {
          console.error('[MealPlan] Failed to persist generated plan:', err.message);
          // Non-fatal — still return the plan even if DB write fails
        }

        // Cache in memory
        setCachedPlan(userId, weekStart, JSON.parse(JSON.stringify(plan)));

        return { plan, fromCache: false, status: 'active' };
      }

      // Step 9: All attempts failed — fallback
      console.warn('[MealPlan] All generation attempts failed, returning fallback');
      const fallback = await generateFallbackMealPlan(calorieTarget, weekStart, dbFoods);
      fallback.calorie_target = calorieTarget;
      fallback.generated_at = new Date().toISOString();
      fallback.llm_model = 'template-fallback';

      // Persist fallback
      try {
        await upsertPlan(userId, weekStart, fallback, 'fallback');
      } catch (err) {
        console.error('[MealPlan] Failed to persist fallback plan:', err.message);
      }

      return { plan: fallback, fromCache: false, status: 'fallback' };
    }
    ```

## Threat Model

### Trust Boundaries

| Boundary | Description |
|----------|-------------|
| LLM API → Service | Untrusted LLM output crosses into service — may contain hallucinated or malformed data |
| DB → Repository | Trusted data path (food database is seeded/admin-controlled) |
| Cache → Service | Trusted data path (cache is written by our service only) |

### STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-01 | **S**poofing | N/A (service layer) | accept | Auth is handled by JWT middleware in the controller layer (Phase 21). Service has no auth responsibility. |
| T-19-02 | **T**ampering | LLM prompt injection via profile fields | mitigate | Profile fields (fitnessGoal, gender) could contain injection text. The prompt template uses these as interpolated strings, not as code. The template `{{fitnessGoal}}` is a single value — escaping the template rendering prevents injection. However, `buildPrompt` already uses `String(value ?? '')` which neutralizes injection. Additional: log all profile values at `console.debug('[MealPlan] Prompt placeholders', { ... })` before calling LLM for audit. |
| T-19-03 | **T**ampering | LLM hallucinated food names | mitigate | `fuzzyMatchFoodName()` handles via exact → case-insensitive → substring → Levenshtein ≤ 3 cascade. Unmatchable items removed. Food_id is always sourced from database, never from LLM. (Per CONTEXT.md D-02) |
| T-19-04 | **T**ampering | LLM miscalculated calorie values | mitigate | `validateAndFixMealPlan()` recalculates ALL calories server-side from `calories_per_100g × portion / 100`. LLM calorie values are discarded. Discrepancies > 20kcal are logged as warnings. (Per success criterion G-03) |
| T-19-05 | **R**epudiation | N/A (service layer) | accept | Audit logging is not in scope for Phase 19. The service logs warnings for corrections/fallbacks which serve as basic audit trail. |
| T-19-06 | **I**nformation Disclosure | LLM response containing user profile data | accept | The LLM receives profile data (weight, height, age, gender, calorie target) as part of the prompt — this is by design. The LLM response is stored in the meal_plans table. No user data is leaked beyond what's needed for the feature. |
| T-19-07 | **D**enial of Service | Token overflow — food list too large | mitigate | `buildMealPlanPrompt()` truncates food list when total character length exceeds 12,000 (~4K tokens). Truncation maintains category diversity. Multiple fallback LLM models provide redundancy. (Per research R-01) |
| T-19-08 | **D**enial of Service | Correction loop exhausting LLM budget | mitigate | `maxAttempts = 2` (per CONTEXT.md D-03). After 2 failed attempts, the service returns a template-based fallback plan that costs $0 in LLM API calls. |
| T-19-09 | **E**levation of Privilege | N/A (service layer) | accept | Service layer has no auth or role checks. Privilege escalation is prevented by the controller's JWT middleware (Phase 21). Service functions receive `userId` from the authenticated context. |
