    Create `backend/src/services/llm.service.js` following the existing service pattern (exported async functions, ValidationError from '../utils/errors.js').

    The file must implement:

    ```javascript
    import OpenAI from 'openai';
    import fs from 'fs';
    import path from 'path';
    import { fileURLToPath } from 'url';
    import { AppError } from '../utils/errors.js';

    const __dirname = path.dirname(fileURLToPath(import.meta.url));
    const PROMPTS_DIR = path.resolve(__dirname, '../../prompts');
    const OPENROUTER_BASE_URL = process.env.OPENROUTER_BASE_URL || 'https://openrouter.ai/api/v1';
    const APP_URL = process.env.FRONTEND_URL || 'http://localhost:3001';

    // Validate OPENROUTER_API_KEY at module load time (D-05 startup validation)
    const API_KEY = process.env.OPENROUTER_API_KEY;
    if (!API_KEY) {
      console.error('FATAL: OPENROUTER_API_KEY is not set. LLM features will not work.');
      console.error('Set OPENROUTER_API_KEY in backend/.env or environment.');
    }

    // Create OpenAI client configured for OpenRouter (per D-01 and RESEARCH.md)
    // Uses baseURL override to point at OpenRouter instead of OpenAI
    let openaiClient = null;
    function getClient() {
      if (!openaiClient && API_KEY) {
        openaiClient = new OpenAI({
          baseURL: OPENROUTER_BASE_URL,
          apiKey: API_KEY,
          defaultHeaders: {
            'HTTP-Referer': APP_URL,
            'X-OpenRouter-Title': 'Fitness_App',
          },
        });
      }
      return openaiClient;
    }

    // Configuration
    const CONFIG = {
      model: process.env.LLM_MODEL || 'nvidia/nemotron-nano-30b-a3b',  // D-01
      temperature: 0.2,    // D-02: deterministic JSON
      maxTokens: 2000,     // D-03: comfortable for full week plan
      retryDelayMs: 1000,  // 1 second between retries
    };

    /**
     * Read a prompt template file, substituting {{variable}} placeholders.
     * @param {string} filename - File name in PROMPTS_DIR
     * @param {Object} variables - Key-value pairs for substitution
     * @returns {string} Filled prompt string
     */
    export function buildPrompt(filename, variables) {
      const filePath = path.join(PROMPTS_DIR, filename);
      let template = fs.readFileSync(filePath, 'utf-8');
      for (const [key, value] of Object.entries(variables)) {
        const placeholder = new RegExp(`{{${key}}}`, 'g');
        template = template.replace(placeholder, String(value ?? ''));
      }
      return template;
    }

    /**
     * Build the system prompt with user context for plan generation.
     * @param {Object} profile - User profile (weight_kg, height_cm, age, gender, fitness_goal, activity_level)
     * @param {Array} activityHistory - Recent activity logs (last 14 days)
     * @param {Array} activities - All available activities from DB
     * @param {string} weekStartDate - Monday of target week (YYYY-MM-DD)
     * @returns {string} Filled system prompt
     */
    export function buildSystemPrompt(profile, activityHistory, activities, weekStartDate) {
      const topActivityNames = [...new Set(activityHistory.map(a => a.activity_name))].slice(0, 5).join(', ');
      const historyText = activityHistory.length > 0
        ? activityHistory.map(a => `- ${a.logged_date || a.loggedDate}: ${a.activity_name} (${a.duration_min}min, ${a.intensity})`).join('\n')
        : 'No recent activity history.';
      const activitiesText = activities.map(a => `- ${a.name} (${a.estimated_calories} cal, ~${a.duration_min}min)`).join('\n');

      return buildPrompt('system-prompt.md', {
        weightKg: profile.weight_kg,
        heightCm: profile.height_cm,
        age: profile.age,
        gender: profile.gender,
        fitnessGoal: profile.fitness_goal,
        activityLevel: profile.activity_level || 'sedentary',
        activityHistory: historyText,
        topActivityNames,
        availableActivities: activitiesText,
        weekStartDate,
      });
    }

    /**
     * Call OpenRouter API to generate a weekly plan.
     * @param {string} systemPrompt - Filled system prompt
     * @returns {Promise<Object>} Parsed JSON plan object
     * @throws {AppError} If API key missing, API call fails, or response is empty
     */
    export async function callLlmApi(systemPrompt) {
      const client = getClient();
      if (!client) {
        throw new AppError('LlmConfigError', 'OPENROUTER_API_KEY not configured', 503);
      }

      const response = await client.chat.completions.create({
        model: CONFIG.model,
        messages: [
          { role: 'system', content: systemPrompt },
        ],
        temperature: CONFIG.temperature,
        max_tokens: CONFIG.maxTokens,
      });

      const content = response.choices?.[0]?.message?.content;
      if (!content) {
        throw new AppError('LlmEmptyResponse', 'LLM returned empty response', 502);
      }

      // Parse JSON, stripping markdown code fences if present (common LLM behavior)
      const cleaned = content.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
      return JSON.parse(cleaned);
    }

    /**
     * Generate a weekly plan — the main exported function.
     * Coordinates: profile fetch, activity history, prompt building, API call.
     * @param {Object} deps - Dependency injection container
     * @param {Function} deps.getProfile - (userId) => profile object
     * @param {Function} deps.getActivityHistory - (userId, days) => activity rows
     * @param {Function} deps.getActivities - () => all activity rows
     * @param {number} deps.userId - User ID
     * @param {string} deps.weekStart - Monday date string
     * @returns {Promise<Object>} Generated plan object
     */
    export async function generateWeeklyPlan(deps) {
      const { getProfile, getActivityHistory, getActivities, userId, weekStart } = deps;

      const [profile, history, activities] = await Promise.all([
        getProfile(userId),
        getActivityHistory(userId, 14),
        getActivities(),
      ]);

      if (!profile) {
        throw new AppError('ProfileNotFound', 'User profile not found', 400);
      }

      const prompt = buildSystemPrompt(profile, history, activities, weekStart);
      return callLlmApi(prompt);
    }
    ```

    - Descriptive JSDoc comments on exported functions
    - Dependency injection pattern for `generatePlan` (receives db query functions as params) — enables unit testing without mocking actual DB
