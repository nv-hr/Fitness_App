    **Step C: Create verify script at scripts/verify-supabase-connection.js (per D-06)**
    Create `scripts/verify-supabase-connection.js` with this exact logic:

    ```javascript
    import dotenv from 'dotenv';
    import pg from 'pg';

    dotenv.config({ path: '../backend/.env' });

    const { Pool } = pg;

    const pool = new Pool({
      connectionString: process.env.DATABASE_URL,
      ssl: {
        rejectUnauthorized: false,  // Supabase requires SSL
      },
      connectionTimeoutMillis: 10000,
    });

    async function verifyConnection() {
      try {
        const client = await pool.connect();
        const result = await client.query('SELECT 1 AS connected');
        client.release();
        console.log('Supabase connected successfully');
        console.log('Result:', result.rows[0]);
        await pool.end();
        process.exit(0);
      } catch (error) {
        console.error('Supabase connection failed:', error.message);
        await pool.end();
        process.exit(1);
      }
    }

    verifyConnection();
    ```

    Notes:
    - File uses ES module syntax (import/export) since backend/package.json has `"type": "module"`.
    - Path to .env is `../backend/.env` because the script runs from the `scripts/` directory.
    - Pool config values: connectionTimeoutMillis=10000 (10s timeout per the agent's discretion).
    - SSL rejectUnauthorized: false — Supabase uses self-signed certs on free tier.
    - On success: logs "Supabase connected successfully", exits 0.
    - On failure: logs error message, exits 1.

    Per the agent's discretion: Connection pool config uses min=0, max=1, idleTimeoutMillis=30000
    for the verify script (single-shot connection check, not a long-lived pool).

    Do NOT modify `backend/src/server.js` or `backend/src/config/database.js` — those are rewritten in Phase 10.
