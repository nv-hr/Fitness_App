```javascript
export function normalizeDbError(err) {
  const CODE_MAP = {
    '23505': 'UNIQUE_VIOLATION',
    '23503': 'FOREIGN_KEY_VIOLATION',
    '23502': 'NOT_NULL_VIOLATION',
    '23514': 'CHECK_VIOLATION',
  };
  return {
    code: CODE_MAP[err.code] || err.code || 'UNKNOWN_DATABASE_ERROR',
    message: err.message || 'Database error occurred',
    sqlState: err.code || null,
    table: err.table || null,
    constraint: err.constraint || null,
    detail: err.detail || null,
  };
}
```

IMPORTANT — pg query() returns: `{ rows: [...], rowCount: number }` (NOT the mysql2 destructure `[rows]`)
Every `const [rows] = await pool.query(...)` MUST be rewritten to `const { rows } = await pool.query(...)`

    **Systematic replacements for ALL functions:**

    A. **pg destructuring**: `const [rows] = await pool.query(...)` → `const { rows } = await pool.query(...)`
       - pg returns `{ rows: [...], rowCount: N }` — NOT mysql2's `[rows, fields]`
       - This change is MANDATORY in ALL 11 functions that use pool.query

    B. **Placeholder syntax**: All `?` in SQL queries → `$1`, `$2` etc. (numbered in parameter order)
       - Example: `WHERE user_id = ? AND name LIKE ?` → `WHERE user_id = $1 AND name LIKE $2`

    C. **INSERT + LAST_INSERT_ID pattern**: Replace with `RETURNING *`
       - Current: INSERT ... VALUES (?) —followed by— SELECT * FROM ... WHERE id = LAST_INSERT_ID()
       - New: `INSERT INTO ... VALUES ($1, $2, ...) RETURNING *`
       - Return `rows[0]` directly (no need for second query)

    D. **DATE_SUB replacement**: `DATE_SUB(CURDATE(), INTERVAL ? DAY)` → `CURRENT_DATE - $1::interval`
       - In getLogHistory (line 136-140): the entire WHERE clause changes

    E. **ER_DUP_ENTRY**: Remove all `err.code === 'ER_DUP_ENTRY'` checks
       - These checks will be handled by normalizeDbError() in the controller (Task 2)
       - For now, catch blocks should just use AppError with a generic error name
       - The ER_DUP_ENTRY-specific handling moves to the controller

    F. **Boolean handling**: `is_custom = ?` with `[is_custom ? 1 : 0]` values → `is_custom = $1` with plain boolean
       - PostgreSQL has native BOOLEAN type — pass `true`/`false` directly
       - In `countFoods` and `findByCategory`: remove the ternary `? 1 : 0`
       - The caller in backend/src/services/food.service.js passes `{ is_custom: false }` — this will work directly with pg
