Create a generic `useMonthData` hook that accepts a month date and a fetch function, then fetches 5-6 weeks of plan data in parallel:

```js
import { useMemo } from 'react';
import { useQueries } from '@tanstack/react-query';
import { format } from 'date-fns';
import { getWeekStartsForMonth, computeDayStatus } from '../calendarUtils.js';

/**
 * Generic hook for fetching month-range plan data.
 * 
 * @param {Date} date — The displayed month date (first day of month or any date in month)
 * @param {function} fetchWeekFn — Function that accepts a weekStart date string (YYYY-MM-DD)
 *   and returns a Promise<{ plan: { days: [{ date: string, completed: boolean, ... }] } }>
 *   Shape matches existing weeklyPlanApi.getWeeklyPlan() response.
 * @returns {{ dayStatusMap: Map<string, string>, loading: boolean, error: Error|null, refetch: function }}
 */
export function useMonthData(date, fetchWeekFn) {
  // 1. Compute week starts for the month range
  const weekStarts = useMemo(() => getWeekStartsForMonth(date), [date]);
  
  // 2. Create query keys and fetch configs for each week
  const weekStartStrings = useMemo(() => 
    weekStarts.map(d => format(d, 'yyyy-MM-dd')),
    [weekStarts]
  );

  const queries = useMemo(() => 
    weekStartStrings.map(weekStart => ({
      queryKey: ['calendarMonthData', weekStart],
      queryFn: () => fetchWeekFn(weekStart),
      staleTime: 5 * 60 * 1000, // 5 min — matches existing pattern
      retry: 1,
    })),
    [weekStartStrings, fetchWeekFn]
  );

  // 3. Execute all queries in parallel via useQueries
  const results = useQueries({ queries });

  // 4. Aggregate results into dayStatusMap
  const dayStatusMap = useMemo(() => {
    const map = new Map();
    const today = new Date();
    
    results.forEach(result => {
      if (!result.data?.plan?.days) return;
      result.data.plan.days.forEach(planDay => {
        const dateStr = planDay.date; // 'YYYY-MM-DD'
        const isPast = planDay.date < format(today, 'yyyy-MM-dd');
        const status = computeDayStatus(dateStr, planDay, isPast);
        map.set(dateStr, status);
      });
    });
    
    return map;
  }, [results]);

  // 5. Compute loading and error states
  const loading = results.some(r => r.isLoading);
  const firstError = results.find(r => r.error)?.error || null;

  return { dayStatusMap, loading, error: firstError, refetch: () => results.forEach(r => r.refetch()) };
}
```

Key design decisions (per D-16 and CONTEXT.md):
- Fetch all weeks in parallel via useQueries (TanStack React Query parallel query hook)
- Returns `dayStatusMap` as a `Map<string, DAY_STATUS>` (not plain object) — O(1) lookups ideal for modifier functions
- Cache: uses TanStack React Query's default caching with staleTime 5min (existing pattern)
- Loading: `results.some(r => r.isLoading)` — true until ALL queries resolve or fail
- Error: first error encountered across all queries
- Refetch: calls refetch on all queries
- fetchWeekFn is a prop so activity/meal pages can pass their own API function (e.g., `getWeeklyPlan` or `getDailyMealPlan`)
- useMemo on weekStarts prevents unnecessary recalculation when date hasn't changed

Per D-16: "Month data fetched via 5-6 parallel weekly plan calls (not 28-31 daily) using getWeekStartsForMonth()"
Per D-06: "5-6 weekly plan fetches via Promise.all" — useQueries handles this internally

```jsx
import { useState, useCallback } from 'react';
import { startOfMonth } from 'date-fns';
import MonthNav from './MonthNav.jsx';
import CalendarGrid from './CalendarGrid.jsx';
import DayDetailPanel from './DayDetailPanel.jsx';
import { useResponsive } from '../../shared/hooks/useResponsive.js';

export default function CalendarPageLayout({ dayStatusMap, loading, error, onMonthChange: externalOnMonthChange, children }) {
  // Internal state
  const [currentMonth, setCurrentMonth] = useState(() => startOfMonth(new Date())); // DEFAULTS TO TODAY per CAL-FND-05
  const [selectedDay, setSelectedDay] = useState(null);
  const { isMobile } = useResponsive();

  // Handle month navigation
  const handleMonthChange = useCallback((month) => {
    setCurrentMonth(startOfMonth(month));
    setSelectedDay(null); // Reset selection on month change per spec
    if (externalOnMonthChange) externalOnMonthChange(month);
  }, [externalOnMonthChange]);

  // Handle Today button click
  const handleTodayClick = useCallback(() => {
    const today = new Date();
    setCurrentMonth(startOfMonth(today));
    setSelectedDay(null);
    if (externalOnMonthChange) externalOnMonthChange(today);
  }, [externalOnMonthChange]);

  // Handle day selection
  const handleDaySelect = useCallback((day) => {
    setSelectedDay(day);
  }, []);

  return (
    <div style={{
      display: 'flex',
      flexDirection: 'column',
      gap: '4px',
      maxWidth: isMobile ? '100%' : '600px',
      margin: '0 auto',
    }}>
      <MonthNav
        currentMonth={currentMonth}
        onMonthChange={handleMonthChange}
        onTodayClick={handleTodayClick}
      />
      
      {loading ? (
        <div style={{ textAlign: 'center', padding: '2rem', color: '#9ca3af' }}>
          Loading...
        </div>
      ) : (
        <CalendarGrid
          currentMonth={currentMonth}
          onMonthChange={handleMonthChange}
          selectedDay={selectedDay}
          onDaySelect={handleDaySelect}
          dayStatusMap={dayStatusMap}
        />
      )}
      
      <DayDetailPanel selectedDay={selectedDay}>
        {children}
      </DayDetailPanel>
    </div>
  );
}
```
